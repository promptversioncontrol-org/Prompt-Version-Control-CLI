 263: async function runWatchLoop(cwd: string, sessionId: string, intervalMs: number = 3000) {
 264:   const watchDir = getWatchDir(cwd, sessionId);
 265:   const watchDirName = path.basename(watchDir);
 266: 
 267:   console.log(`👁️  Watching session: ${sessionId}`);
 268:   console.log(`📁 Watch folder: ${watchDirName}`);
 269:   console.log(`⏱️  Checking every ${intervalMs}ms`);
 270:   console.log(`🛑 Stop with: pvc watch stop\n`);
 271: 
 272:   let lastMtime = 0;
 273:   let notFoundCount = 0;
 274: 
 275:   while (true) {
 276:     try {
 277:       const sessionFile = findCodexSessionFile(sessionId);
 278: 
 279:       if (!sessionFile) {
 280:         notFoundCount++;
 281:         if (notFoundCount === 1) {
 282:           console.log("⏳ Waiting for session file to appear...");
 283:         }
 284:         await new Promise(r => setTimeout(r, intervalMs));
 285:         continue;
 286:       }
 287: 
 288:       if (notFoundCount > 0) {
 289:         console.log(`✅ Session file found: ${sessionFile}\n`);
 290:         notFoundCount = 0;
 291:       }
 292: 
 293:       const stat = statSync(sessionFile);
 294: 
 295:       if (stat.mtimeMs > lastMtime) {
 296:         lastMtime = stat.mtimeMs;
 297:         console.log(`\n🔔 New activity detected at ${new Date().toLocaleTimeString()}`);
 298:         await updateReports(cwd, sessionId, sessionFile);
 299:         console.log("");
 300:       }
 301:     } catch (err) {
 302:       console.error(`❌ Error: ${(err as Error).message}`);
 303:     }
 304: 
 305:     await new Promise(r => setTimeout(r, intervalMs));
 306:   }
 307: }
